# AeroServe PaaS

A lightweight Platform-as-a-Service (PaaS) designed to serve machine learning models efficiently in production environments.

## Quickstart Local Setup

Start the entire local infrastructure in under 5 commands[cite: 1]:

```bash
# 1. Clone the repository and navigate into it
git clone <your-repo-url> && cd aeroserve

# 2. Build and start the services (Gateway, Redis, Postgres)
docker compose up --build -d

# 3. Verify all services are healthy
docker compose ps

# 4. Test the API Gateway liveness endpoint
curl http://localhost:8000/health

import threading

def fire_request():
    url = "http://localhost:8000/predict"
    payload = {
        "model_name": "resnet50_classifier",
        "priority": "background",
        "payload": {"data": "burst_test"}
    }
    while True: # Keep firing continuously
        try:
            requests.post(url, json=payload)
        except Exception:
            pass
        time.sleep(0.1) # Small delay to prevent crashing your own machine

print("Starting sustained load test (Press Ctrl+C to stop)...")

for _ in range(20):

import argparse
import json
import os
import shutil
import sys
import time
import urllib.request
import urllib.error

# Ensure imports work from the root directory
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from tools.profiler import run_profiling

def main():
    parser = argparse.ArgumentParser(prog="aeroserve", description="AeroServe PaaS CLI Tooling")
    subparsers = parser.add_subparsers(dest="command", required=True)

    submit_cmd = subparsers.add_parser("submit", help="Profile and register a local model package")
    submit_cmd.add_argument("--model-dir", type=str, required=True, help="Path to folder containing processor.py & test_inputs.json")
    submit_cmd.add_argument("--name", type=str, required=True, help="Registry name for the model")
    submit_cmd.add_argument("--framework", type=str, default="transformers", choices=["transformers", "pytorch", "tensorflow", "onnx", "custom"])
    submit_cmd.add_argument("--sla", type=float, default=None, help="Latency target threshold in milliseconds")
    submit_cmd.add_argument("--processor-class", type=str, default=None, help="Exact class name to load from processor.py (Optional)")

    args = parser.parse_args()

    if args.command == "submit":
        abs_model_dir = os.path.abspath(args.model_dir)
        inputs_file = os.path.join(abs_model_dir, "test_inputs.json")
        processor_file = os.path.join(abs_model_dir, "processor.py")

        if not os.path.isfile(processor_file):
            print(f"[-] Missing processor implementation: {processor_file}")
            sys.exit(1)

        if not os.path.isfile(inputs_file):
            print(f"[-] Missing test inputs: {inputs_file}")
            sys.exit(1)

        with open(inputs_file, "r") as f:
            try:
                test_records = json.load(f)
                if not isinstance(test_records, list) or len(test_records) == 0:
                    raise ValueError("Must contain a non-empty JSON array")
            except Exception as e:
                print(f"[-] Invalid test_inputs.json format: {e}")
                sys.exit(1)

        # Run profiling securely
        generated_config = run_profiling(
            model_name=args.name,
            source_dir=abs_model_dir,
            framework=args.framework,
            sla_target_ms=args.sla,
            test_inputs=test_records,
            output_dir=abs_model_dir,
            target_class_name=args.processor_class
        )

        # Atomic Deployment
        live_models_dir = os.path.abspath("models")
        temp_model_dir = os.path.join(live_models_dir, f".tmp_{args.name}")
        final_model_dir = os.path.join(live_models_dir, args.name)

        if os.path.exists(temp_model_dir):
            shutil.rmtree(temp_model_dir)
        os.makedirs(temp_model_dir, exist_ok=True)

        print("\n[*] Transferring model artifacts...")
        for item in os.listdir(abs_model_dir):
            src_item = os.path.join(abs_model_dir, item)
            dst_item = os.path.join(temp_model_dir, item)
            if os.path.isfile(src_item):
                shutil.copy2(src_item, dst_item)

        # Atomic Swap: Instantly replaces the old model or provisions a new one safely
        if os.path.exists(final_model_dir):
            shutil.rmtree(final_model_dir)
        os.rename(temp_model_dir, final_model_dir)
        print("[+] Artifacts deployed safely via atomic swap.")

        # Verification
        print("[*] Waiting for AeroServe gateway and workers to acknowledge the model...")
        max_retries = 30
        is_ready = False
        
        for attempt in range(max_retries):
            try:
                # We will ensure the Gateway has a /health endpoint that checks active models later
                req = urllib.request.Request(f"http://localhost:8000/health")
                with urllib.request.urlopen(req, timeout=2) as response:
                    if response.status == 200:
                        health_data = json.loads(response.read().decode('utf-8'))
                        
                        # Verify our specific model is in the active registry
                        if args.name in health_data.get("models_loaded", []):
                            is_ready = True
                            break
            except urllib.error.URLError:
                pass
            
            time.sleep(1)
            print(".", end="", flush=True)

        if not is_ready:
            print("\n[-] Deployment warning: Files deployed, but Gateway healthcheck timed out.")
            sys.exit(1)

        print(f"\n[+] Zero-Touch Deployment Complete & Verified!")
        print(f"    - Endpoint available at: POST /predict")
        print(f"    - Model Name: {args.name}")
        print(f"    - Max Batch Size: {generated_config['batching']['max_batch_size']}")

if __name__ == "__main__":
    main()

import asyncio

import time

URL = "http://localhost:8000/predict"

    "model_name": "iris_model",
    "priority": "standard",
    "user_tier": "user",
    "payload": {"features": [5.1, 3.5, 1.4, 0.2]}
}
DURATION_SECONDS = 40

async def send_requests(session):
    requests_sent = 0
    end_time = time.time() + DURATION_SECONDS
    while time.time() < end_time:
        try:
            async with session.post(URL, json=PAYLOAD) as response:
                await response.read()
                requests_sent += 1
        except Exception:
            pass
    return requests_sent

async def main():
    print(f"ðŸš€ Firing asynchronous blast for {DURATION_SECONDS} seconds...")

        tasks = [send_requests(session) for _ in range(200)]

import os
import sys
import time
import math
import logging
import subprocess
import requests
import json

PROJECT_ROOT = os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))
sys.path.insert(0, PROJECT_ROOT)

from config import settings

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] [Autoscaler]: %(message)s")
logger = logging.getLogger("AeroServe.Autoscaler")

PROMETHEUS_URL = "http://prometheus:9090"
POLL_INTERVAL_SEC = 5

def get_dynamic_scaling_limits():
    """Scans all model_config.json files for zero-touch scaling limits."""
    base_min_workers = 1
    aggregate_max_workers = 1
    avg_target_queue = 15
    
    models_dir = os.path.join(PROJECT_ROOT, "models")
    if not os.path.exists(models_dir):
        return base_min_workers, aggregate_max_workers, avg_target_queue
        
    targets = []
    for folder in os.listdir(models_dir):
        if folder.startswith(".tmp_"): continue
        
        config_path = os.path.join(models_dir, folder, "model_config.json")
        if os.path.exists(config_path):
            try:
                with open(config_path, "r") as f:
                    config = json.load(f)
                    autoscaling = config.get("autoscaling", {})
                    
                    m_max = autoscaling.get("max_workers", 1)
                    if m_max > aggregate_max_workers:
                        aggregate_max_workers = m_max
                        
                    targets.append(autoscaling.get("target_queue_per_worker", 15))
            except Exception as e:
                logger.error(f"Failed to read scaling config from {folder}: {e}")
                
    if targets:
        avg_target_queue = sum(targets) // len(targets)
        
    return base_min_workers, aggregate_max_workers, avg_target_queue

class QueueAutoscaler:
    def __init__(self):
        self.min_w, self.max_w, self.target_q = get_dynamic_scaling_limits()
        self.current_workers = self.min_w
        self.last_scale_time = 0.0

    def query_prometheus(self, query: str) -> float:
        try:
            response = requests.get(f"{PROMETHEUS_URL}/api/v1/query", params={"query": query}, timeout=3)
            response.raise_for_status()
            results = response.json().get("data", {}).get("result", [])
            if results:
                return float(results[0]["value"][1])
            return 0.0
        except Exception:
            return 0.0

    def execute_scale(self, target_workers: int, action: str):
        logger.info(f"Action: {action} | Scaling workers from {self.current_workers} -> {target_workers}")
        try:
            subprocess.run(
                ["docker", "compose", "up", "-d", "--scale", f"worker={target_workers}"],
                capture_output=True, text=True, check=True, cwd=PROJECT_ROOT 
            )
            self.current_workers = target_workers
            self.last_scale_time = time.time()
            logger.info(f"Scale successful. New worker count: {target_workers}")
        except subprocess.CalledProcessError as e:
            logger.error(f"Scale command failed! STDERR: {e.stderr}")

    def evaluate(self):
        now = time.time()
        time_since_last_scale = now - self.last_scale_time

        self.min_w, self.max_w, self.target_q = get_dynamic_scaling_limits()

        total_queue_depth = self.query_prometheus("sum(aeroserve_queue_depth)")
        oldest_age = self.query_prometheus("max(aeroserve_oldest_request_age_seconds)")

        depth_desired = math.ceil(total_queue_depth / self.target_q) if self.target_q > 0 else self.min_w
        
        age_desired = 0
        if oldest_age > 2.0: 
            age_desired = self.current_workers + 1
            
        raw_desired = max(depth_desired, age_desired)
        desired_workers = max(self.min_w, min(raw_desired, self.max_w))

        # FIX: Using the uppercase configuration settings
        if desired_workers > self.current_workers:
            if time_since_last_scale >= settings.SCALE_UP_COOLDOWN_SEC:
                logger.info("UPWARD PRESSURE DETECTED.")
                self.execute_scale(desired_workers, "SCALE UP")
            else:
                cooldown_left = settings.SCALE_UP_COOLDOWN_SEC - time_since_last_scale
                logger.info(f"Scale up in cooldown ({cooldown_left:.0f}s left)")

        elif desired_workers < self.current_workers:
            current_capacity = self.current_workers * self.target_q
            hysteresis_threshold = current_capacity * settings.SCALE_DOWN_HYSTERESIS_FACTOR

            if total_queue_depth < hysteresis_threshold and oldest_age == 0:
                if time_since_last_scale >= settings.SCALE_DOWN_COOLDOWN_SEC:
                    logger.info("DOWNWARD PRESSURE DETECTED.")
                    self.execute_scale(desired_workers, "SCALE DOWN")
                else:
                    cooldown_left = settings.SCALE_DOWN_COOLDOWN_SEC - time_since_last_scale
                    logger.info(f"Scale down in cooldown ({cooldown_left:.0f}s left)")

    def run(self):
        logger.info("AeroServe Queue-Driven Autoscaler Started.")
        while True:
            self.evaluate()
            time.sleep(POLL_INTERVAL_SEC)

if __name__ == "__main__":
    autoscaler = QueueAutoscaler()
    try:
        autoscaler.run()
    except KeyboardInterrupt:
        logger.info("Autoscaler shut down by user.")

def load_model_registry():
    """Reloads the models in memory so zero-touch updates are recognized"""
    if not os.path.exists(MODELS_DIR):
        print(f"[Gateway] Warning: Models directory '{MODELS_DIR}' not found.")
        return
    
    current_models = {}
    for model_folder in os.listdir(MODELS_DIR):
        # Skip temporary folders created by the CLI during atomic swap
        if model_folder.startswith(".tmp_"):
            continue
            
        config_path = os.path.join(MODELS_DIR, model_folder, "model_config.json")
        if os.path.exists(config_path):
            try:
                with open(config_path, "r") as f:
                    config = json.load(f)
                    model_name = config["model_name"]
                    current_models[model_name] = config
                    print(f"[Gateway] Registered Model: {model_name}")
            except Exception as e:
                print(f"[Gateway] Failed to load config for {model_folder}: {e}")
                
    # Update global registry atomically
    global model_registry
    model_registry = current_models

async def update_prometheus_metrics(redis_client: redis.Redis):
    while True:
        try:
            # Periodically reload models to support the CLI zero-touch deployment
            load_model_registry()
            
            for priority in KNOWN_PRIORITIES:
                queue_name = f"queue:{priority}"
                depth = await redis_client.llen(queue_name)
                AEROSERVE_QUEUE_DEPTH.labels(priority=priority).set(depth)
                
                if depth > 0:
                    oldest_req_bytes = await redis_client.lrange(queue_name, 0, 0)
                    if oldest_req_bytes:
                        req_data = json.loads(oldest_req_bytes[0])
                        created_at = req_data.get("created_at", time.time())
                        AEROSERVE_OLDEST_AGE.labels(priority=priority).set(time.time() - created_at)
                else:
                    AEROSERVE_OLDEST_AGE.labels(priority=priority).set(0.0)
        except Exception as e:
            print(f"[Gateway] Metrics/Registry error: {e}")
        await asyncio.sleep(2)

async def log_request_to_db(pool, req_id: str, model: str, priority: str, req_status: str, result: dict | None, latency: float):
    # Upsert query handles both the initial "pending" log and the final "success/error" update
    query = """
        INSERT INTO inference_logs (request_id, model_name, priority, status, inference_result, latency_ms)
        VALUES ($1, $2, $3, $4, $5, $6)
        ON CONFLICT (request_id) 
        DO UPDATE SET 
            status = EXCLUDED.status, 
            inference_result = EXCLUDED.inference_result, 
            latency_ms = EXCLUDED.latency_ms
    """
    try:
        async with pool.acquire() as conn:
            await conn.execute(query, req_id, model, priority, req_status, json.dumps(result) if result else None, latency)
        AEROSERVE_DB_WRITES.labels(status="success").inc()
    except Exception as e:
        AEROSERVE_DB_WRITES.labels(status="error").inc()
        print(f"[Gateway] DB Logging Error for {req_id}: {e}")

@app.post("/predict", response_model=PredictResponse, tags=["Inference Routing"])
async def predict(
    request_data: PredictRequest,
    background_tasks: BackgroundTasks,
    request: Request,
    redis_client: redis.Redis = Depends(get_redis_client)
):
    model_name = request_data.model_name
    
    # FIX: Prevent phantom requests. Act as a strict bouncer.
    if model_name not in model_registry:
        raise HTTPException(status_code=404, detail=f"Model '{model_name}' not found or not loaded.")

    priority_val = getattr(request_data.priority, 'value', request_data.priority)
    queue_name = f"queue:{priority_val}"
    
    slas = model_registry[model_name].get("slas", {})
    dynamic_timeout = slas.get("priority_timeouts_s", {}).get(priority_val, slas.get("max_queue_wait_s", 45.0))

    request_id = str(uuid.uuid4())
    timestamp = time.time()
    
    envelope = {
        "request_id": request_id,
        "model_name": model_name,
        "created_at": timestamp,                    
        "priority": priority_val,
        "user_tier": getattr(request_data.user_tier, 'value', request_data.user_tier),
        "payload": request_data.payload.model_dump() # Pass the dict, not the Pydantic object
    }
    
    pubsub = redis_client.pubsub()
    response_channel = f"response:{request_id}"
    await pubsub.subscribe(response_channel)
    
    # FIX: Ensure no requests are lost to DB if queue crashes
    background_tasks.add_task(log_request_to_db, request.app.state.db_pool, request_id, model_name, priority_val, "pending", None, 0.0)
    
    AEROSERVE_IN_FLIGHT.labels(priority=priority_val).inc()
    
    try:
        try:
            await redis_client.rpush(queue_name, json.dumps(envelope))
            AEROSERVE_REQUEST_COUNT.labels(priority=priority_val, model_name=model_name).inc()
        except Exception as exc:
            raise HTTPException(status_code=503, detail="Queue backend unavailable.")
            
        try:
            async with asyncio.timeout(dynamic_timeout): 
                async for message in pubsub.listen():
                    if message["type"] == "message":
                        result = json.loads(message["data"])
                        latency_s = time.time() - timestamp
                        latency_ms = latency_s * 1000
                        
                        AEROSERVE_LATENCY.labels(priority=priority_val).observe(latency_s)
                        
                        if result.get("status") == "error":
                            AEROSERVE_INFERENCE_STATUS.labels(priority=priority_val, status="worker_error").inc()
                            background_tasks.add_task(log_request_to_db, request.app.state.db_pool, request_id, model_name, priority_val, "error", {"error": result.get("error")}, latency_ms)
                            raise HTTPException(status_code=500, detail=result.get("error"))
                        
                        AEROSERVE_INFERENCE_STATUS.labels(priority=priority_val, status="success").inc()
                        background_tasks.add_task(log_request_to_db, request.app.state.db_pool, request_id, model_name, priority_val, "success", result.get("data"), latency_ms)
                        
                        return PredictResponse(
                            request_id=request_id,
                            status="completed",
                            queue=queue_name,
                            timestamp=timestamp,
                            inference_result=result.get("data")
                        )
        except asyncio.TimeoutError:
            latency_s = time.time() - timestamp
            latency_ms = latency_s * 1000
            AEROSERVE_LATENCY.labels(priority=priority_val).observe(latency_s)
            AEROSERVE_INFERENCE_STATUS.labels(priority=priority_val, status="timeout").inc()
            
            background_tasks.add_task(log_request_to_db, request.app.state.db_pool, request_id, model_name, priority_val, "timeout", None, latency_ms)
            raise HTTPException(status_code=504, detail=f"Worker timeout for SLA: {dynamic_timeout}s")
        finally:
            await pubsub.unsubscribe(response_channel)
            await pubsub.close() 
    finally:
        AEROSERVE_IN_FLIGHT.labels(priority=priority_val).dec()

import enum
import base64
from typing import Any
from pydantic import BaseModel, Field, field_validator,model_validator

class PriorityTier(str, enum.Enum):
    PREMIUM = "premium"
    STANDARD = "standard"
    BACKGROUND = "background"

class UserTier(str, enum.Enum):
    ADMIN = "admin"
    USER = "user"
    GUEST = "guest"

class InferencePayload(BaseModel):
    text: str | None = Field(default=None, description="Raw text string for NLP models")
    image_b64: str | None = Field(default=None, description="Base64 encoded image string for Computer Vision")
    features: list[float] | None = Field(default=None, description="1D array of numerical features for tabular models")
    tensors: list[Any] | None = Field(default=None, description="Multi-dimensional array structures for raw tensor input")

    @field_validator('image_b64')
    @classmethod
    def validate_image_b64(cls, v: str | None) -> str | None:
        if v is not None:
            try:
                # Validate it is actually a decodable base64 string
                base64.b64decode(v, validate=True)
            except Exception:
                raise ValueError("image_b64 must be a valid base64 encoded string")
        return v

    @model_validator(mode='after')
    def check_at_least_one_input(self):
        if not any([self.text, self.image_b64 is not None, self.features, self.tensors]):
            raise ValueError("Payload must contain at least one input type (text, image_b64, features, or tensors).")
        return self

class PredictRequest(BaseModel):
    model_name: str = Field(..., description="Target model identifier registered in AeroServe")
    payload: InferencePayload = Field(..., description="Strictly typed input data payload")
    priority: PriorityTier = Field(
        default=PriorityTier.STANDARD, 
        description="Inference priority tier: premium, standard, or background"
    )
    user_tier: UserTier = Field(
        default=UserTier.GUEST,
        description="Access tier of the requesting user"
    )

class PredictResponse(BaseModel):
    request_id: str = Field(..., description="Correlation ID for end-to-end tracing")
    status: str = Field(default="enqueued", description="Lifecycle status of the request")
    queue: str = Field(..., description="Target Redis queue where the job is stored")
    timestamp: float = Field(..., description="UNIX timestamp when the request was ingested")
    inference_result: dict[str, Any] | None = Field(default=None, description="The final prediction output from the worker")

class HealthResponse(BaseModel):
    status: str
    redis_connected: bool
    models_loaded: list[str] = Field(default_factory=list, description="List of actively registered models")

from abc import ABC, abstractmethod
from typing import Any


class AeroServeProcessor(ABC):
    """
    Base interface for all AeroServe processors.

    Every processor must implement execute(), which accepts
    a validated inference payload and returns a dictionary
    containing the inference result.
    """

    @abstractmethod
    def execute(self, payload: dict[str, Any]) -> dict[str, Any]:
        """Execute inference and return a JSON-serializable dictionary."""
        raise NotImplementedError


class AeroServeManagedProcessor(AeroServeProcessor):
    """
    Framework-managed inference pipeline.

    AeroServe controls the execution sequence:
        preprocess -> predict -> postprocess

    Developers implement the three individual stages.
    """

    @abstractmethod
    def preprocess(self, payload: dict[str, Any]) -> Any:
        """Transform the validated payload into tensors, arrays, or model inputs."""
        raise NotImplementedError

    @abstractmethod
    def predict(self, model_input: Any) -> Any:
        """Execute the model's forward pass."""
        raise NotImplementedError

    @abstractmethod
    def postprocess(self, inference_output: Any) -> dict[str, Any]:
        """Convert model outputs into a JSON-serializable dictionary."""
        raise NotImplementedError

    def execute(self, payload: dict[str, Any]) -> dict[str, Any]:
        """Run the complete managed inference pipeline."""
        features = self.preprocess(payload)
        raw_output = self.predict(features)
        result = self.postprocess(raw_output)

        if not isinstance(result, dict):
            raise TypeError(
                "postprocess() must return a dictionary."
            )

        return result


class AeroServeCustomProcessor(AeroServeProcessor):
    """
    Developer-managed inference pipeline.

    Developers implement execute() themselves when their
    inference workflow requires custom orchestration.
    """

    @abstractmethod
    def execute(self, payload: dict[str, Any]) -> dict[str, Any]:
        """Execute the full inference cycle and return a result dictionary."""
        raise NotImplementedError

import os
import joblib
import numpy as np
import time
from typing import Any, Dict, List

# Import the base class expected by AeroServe
from models.aeroserve_preprocessor import AeroServeProcessor

class IrisProcessor(AeroServeProcessor):
    # Enable AeroServe's dynamic batching engine
    supports_batching = True

    def __init__(self):
        # Resolve path dynamically so it works inside the Docker worker volume
        current_dir = os.path.dirname(os.path.abspath(__file__))
        artifact_path = os.path.join(current_dir, "iris_pipeline.joblib")
        
        print(f"Loading Iris Model and Scaler from {artifact_path}...")
        artifacts = joblib.load(artifact_path)
        self.model = artifacts['model']
        self.scaler = artifacts['scaler']

    def execute(self, payload: Dict[str, Any]) -> Dict[str, Any]:
        """Handles a single inference request."""
        features = payload.get("features")
        
        if not features or len(features) != 4:
            return {"status": "error", "error": "Payload must contain a 'features' array of exactly 4 numeric values."}
        
        # Reshape for sklearn (1 sample, 4 features)
        X = np.array([features])
        X_scaled = self.scaler.transform(X)
        prediction = self.model.predict(X_scaled)[0]
        
        return {"status": "success", "prediction": str(prediction)}

    def execute_batch(self, payloads: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        """Handles dynamically batched requests for high throughput."""
        time.sleep(10)
        X_list = []
        valid_indices = []
        
        # Initialize default error responses
        results = [{"status": "error", "error": "Invalid features array"} for _ in payloads]
        
        for idx, p in enumerate(payloads):
            feats = p.get("features")
            if feats and len(feats) == 4:
                X_list.append(feats)
                valid_indices.append(idx)
        
        # Perform vectorized inference on valid payloads
        if X_list:
            X_array = np.array(X_list)
            X_scaled = self.scaler.transform(X_array)
            predictions = self.model.predict(X_scaled)
            
            # Map predictions back to their original positions in the batch
            for i, pred_val in enumerate(predictions):
                original_idx = valid_indices[i]
                results[original_idx] = {"status": "success", "prediction": str(pred_val)}
                
        return results

logger = logging.getLogger("AeroServe.Scheduler")

QUEUE_DEPTH = Gauge('aeroserve_queue_depth', 'Current requests in queue', ['priority'])
OLDEST_AGE = Gauge('aeroserve_oldest_request_age_seconds', 'Age of oldest request', ['priority'])
AEROSERVE_TARGET_WORKERS = Gauge("aeroserve_target_workers", "Number of worker replicas requested by the autoscaler")

async def metrics_publisher_loop(redis_client: redis_async.Redis):
    queues = ["premium", "standard", "background"]
    logger.info("Metrics publisher loop started on port 8001")
    
    while True:
        try:
            for q_name in queues:
                queue_key = f"queue:{q_name}"
                depth = await redis_client.llen(queue_key)
                QUEUE_DEPTH.labels(priority=q_name).set(depth)
                
                if depth > 0:
                    oldest_raw = await redis_client.lindex(queue_key, -1)
                    if oldest_raw:
                        task = json.loads(oldest_raw)
                        age = time.time() - task["created_at"]
                        OLDEST_AGE.labels(priority=q_name).set(age)
                else:
                    OLDEST_AGE.labels(priority=q_name).set(0)
                    
        except Exception as e:
            logger.error(f"Error publishing metrics: {e}")
            
        await asyncio.sleep(2)

async def health_check_loop(redis_client: redis_async.Redis, cb: WorkerCircuitBreaker, rerouter: RequestRerouter):

import json
import logging
from typing import List, Dict, Any
from redis.asyncio import Redis

logger = logging.getLogger("AeroServe.Scheduler.Rerouter")


class RequestRerouter:
    """Detects requests assigned to dead/unhealthy workers and requeues them safely."""

    def __init__(self, redis: Redis):
        self.redis = redis

    def _processing_queue(self, worker_id: str) -> str:
        return f"queue:processing:{worker_id}"

    async def recover_orphaned_requests(self, dead_worker_id: str) -> int:
        processing_queue = self._processing_queue(dead_worker_id)
        requeued_count = 0

        while True:
            # Atomic fetch from dead worker's processing queue
            raw_payload = await self.redis.rpop(processing_queue)
            if not raw_payload:
                break

            payload_str = raw_payload.decode("utf-8") if isinstance(raw_payload, bytes) else raw_payload
            try:
                request_data: Dict[str, Any] = json.loads(payload_str)
            except json.JSONDecodeError:
                logger.error(f"Malformed payload dropped during recovery: {payload_str}")
                continue

            priority = request_data.get("priority", "standard").lower()
            target_queue = f"queue:{priority}"

            # Re-enqueue at the front (LPUSH) so orphaned requests retain priority
            request_data["requeued"] = True
            await self.redis.lpush(target_queue, json.dumps(request_data))
            requeued_count += 1

        if requeued_count > 0:
            logger.warning(
                f"Recovered and requeued {requeued_count} orphaned request(s) from worker {dead_worker_id}."
            )
        return requeued_count

import enum
import time
from typing import Optional
from redis.asyncio import Redis


class WorkerState(str, enum.Enum):
    HEALTHY = "healthy"
    UNHEALTHY = "unhealthy"
    DRAINING = "draining"


class CircuitBreakerConfig:
    def __init__(
        self,
        heartbeat_ttl_sec: int = 5,
        consecutive_success_threshold: int = 3,
        failure_threshold: int = 2,
    ):
        self.heartbeat_ttl_sec = heartbeat_ttl_sec
        self.consecutive_success_threshold = consecutive_success_threshold
        self.failure_threshold = failure_threshold


class WorkerCircuitBreaker:
    """Manages the lifecycle and state transitions of individual workers in Redis."""

    def __init__(self, redis: Redis, config: Optional[CircuitBreakerConfig] = None):
        self.redis = redis
        self.config = config or CircuitBreakerConfig()

    def _state_key(self, worker_id: str) -> str:
        return f"worker:{worker_id}:state"

    def _heartbeat_key(self, worker_id: str) -> str:
        return f"worker:heartbeat:{worker_id}"

    def _success_counter_key(self, worker_id: str) -> str:
        return f"worker:{worker_id}:consecutive_success"

    def _failure_counter_key(self, worker_id: str) -> str:
        return f"worker:{worker_id}:consecutive_failures"

    async def get_state(self, worker_id: str) -> WorkerState:
        raw = await self.redis.get(self._state_key(worker_id))
        if raw is None:
            return WorkerState.HEALTHY
        decoded = raw.decode("utf-8") if isinstance(raw, bytes) else raw
        return WorkerState(decoded)

    async def set_state(self, worker_id: str, new_state: WorkerState) -> None:
        await self.redis.set(self._state_key(worker_id), new_state.value)
        if new_state == WorkerState.HEALTHY:
            await self.redis.sadd("active_workers:healthy", worker_id)
            await self.redis.srem("active_workers:unhealthy", worker_id)
            await self.redis.srem("active_workers:draining", worker_id)
        elif new_state == WorkerState.UNHEALTHY:
            await self.redis.srem("active_workers:healthy", worker_id)
            await self.redis.sadd("active_workers:unhealthy", worker_id)
            await self.redis.srem("active_workers:draining", worker_id)
        elif new_state == WorkerState.DRAINING:
            await self.redis.srem("active_workers:healthy", worker_id)
            await self.redis.srem("active_workers:unhealthy", worker_id)
            await self.redis.sadd("active_workers:draining", worker_id)

    async def register_worker(self, worker_id: str) -> None:
        await self.redis.sadd("registered_workers", worker_id)
        await self.set_state(worker_id, WorkerState.HEALTHY)

    async def record_heartbeat_failure(self, worker_id: str) -> WorkerState:
        failures = await self.redis.incr(self._failure_counter_key(worker_id))
        await self.redis.delete(self._success_counter_key(worker_id))

        current_state = await self.get_state(worker_id)
        if failures >= self.config.failure_threshold and current_state != WorkerState.UNHEALTHY:
            await self.set_state(worker_id, WorkerState.UNHEALTHY)
            return WorkerState.UNHEALTHY
        return current_state

    async def record_heartbeat_success(self, worker_id: str) -> WorkerState:
        await self.redis.delete(self._failure_counter_key(worker_id))
        current_state = await self.get_state(worker_id)

        if current_state == WorkerState.UNHEALTHY:
            successes = await self.redis.incr(self._success_counter_key(worker_id))
            if successes >= self.config.consecutive_success_threshold:
                await self.set_state(worker_id, WorkerState.HEALTHY)
                await self.redis.delete(self._success_counter_key(worker_id))
                return WorkerState.HEALTHY
            else:
                await self.set_state(worker_id, WorkerState.DRAINING)
                return WorkerState.DRAINING

        elif current_state == WorkerState.DRAINING:
            successes = await self.redis.incr(self._success_counter_key(worker_id))
            if successes >= self.config.consecutive_success_threshold:
                await self.set_state(worker_id, WorkerState.HEALTHY)
                await self.redis.delete(self._success_counter_key(worker_id))
                return WorkerState.HEALTHY
            return WorkerState.DRAINING

        return WorkerState.HEALTHY

def load_processor_class(model_dir: str, file_name: str = "processor.py", target_class_name: str = None) -> Tuple[Type[AeroServeProcessor], str, str]:
    """Dynamically imports the developer's processor class."""
    target_path = os.path.join(model_dir, file_name)
    if not os.path.isfile(target_path):
        raise FileNotFoundError(f"Missing processor implementation at {target_path}")
    
    spec = importlib.util.spec_from_file_location("dynamic_module", target_path)
    if spec is None or spec.loader is None:
        raise ImportError(f"Cannot load module specification from {target_path}")
    
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    
    matched_classes = []
    for name, obj in inspect.getmembers(module, inspect.isclass):
        if issubclass(obj, AeroServeProcessor) and not inspect.isabstract(obj):
            if target_class_name and name == target_class_name:
                return obj, file_name, name
            matched_classes.append((name, obj))
            
    if not matched_classes:
        for name, obj in inspect.getmembers(module, inspect.isclass):
            if hasattr(obj, "execute") and obj.__module__ == "dynamic_module" and not inspect.isabstract(obj):
                if target_class_name and name == target_class_name:
                    return obj, file_name, name
                matched_classes.append((name, obj))
                
    if target_class_name:
        raise TypeError(f"Class '{target_class_name}' not found in {file_name}")
        
    if not matched_classes:
        raise TypeError(f"No valid concrete subclass of AeroServeProcessor found in {file_name}")
        
    name, selected_class = matched_classes[0]
    return selected_class, file_name, name

import asyncio
import logging
from redis.asyncio import Redis

logger = logging.getLogger("AeroServe.Worker.Heartbeat")

class HeartbeatEmitter:
    """Runs as a lightweight background coroutine inside the worker container."""

    def __init__(self, redis: Redis, worker_id: str, heartbeat_interval_sec: float = 2.0, ttl_sec: int = 5):
        self.redis = redis
        self.worker_id = worker_id
        self.interval = heartbeat_interval_sec
        self.ttl = ttl_sec
        self.is_running = False
        self._task: asyncio.Task = None

    async def start(self):
        self.is_running = True
        await self.redis.sadd("registered_workers", self.worker_id)
        self._task = asyncio.create_task(self._pulse_loop())

    async def _pulse_loop(self):
        heartbeat_key = f"worker:heartbeat:{self.worker_id}"
        while self.is_running:
            try:
                await self.redis.set(heartbeat_key, "1", ex=self.ttl)
            except Exception as e:
                logger.error(f"Failed to publish heartbeat: {e}")
            await asyncio.sleep(self.interval)

    async def stop(self):
        self.is_running = False
        if self._task:
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
        # FIX: Clean up registry on graceful shutdown
        try:
            await self.redis.srem("registered_workers", self.worker_id)
            await self.redis.delete(f"worker:heartbeat:{self.worker_id}")
        except Exception:
            pass

class CircuitState(Enum):
    CLOSED = "CLOSED"       
    OPEN = "OPEN"           
    HALF_OPEN = "HALF_OPEN" 

class CircuitBreaker:
    def __init__(self, failure_threshold: int = settings.CB_FAILURE_THRESHOLD, recovery_timeout: float = 30.0):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = 0.0

    def set_state(self, new_state: CircuitState):
        self.state = new_state
        state_map = {CircuitState.CLOSED: 0, CircuitState.HALF_OPEN: 1, CircuitState.OPEN: 2}
        CB_STATE.labels(worker_id=WORKER_ID).set(state_map[new_state])

    def can_execute(self) -> bool:
        if self.state == CircuitState.CLOSED: return True
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time >= self.recovery_timeout:
                logger.warning("Circuit Breaker entering HALF_OPEN state.")
                self.set_state(CircuitState.HALF_OPEN)
                return True
            return False
        return True

    def record_success(self):
        if self.state != CircuitState.CLOSED:
            logger.info("Worker recovered completely.")
        self.set_state(CircuitState.CLOSED)
        self.failure_count = 0

    def record_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.time()
        if self.state == CircuitState.HALF_OPEN:
            self.set_state(CircuitState.OPEN)
        elif self.state == CircuitState.CLOSED and self.failure_count >= self.failure_threshold:
            logger.error(f"Circuit Breaker TRIPPED to OPEN after {self.failure_count} crashes!")
            self.set_state(CircuitState.OPEN)

class DynamicModelRegistry:
    def __init__(self, models_dir="/app/models"):
        self.models_dir = models_dir
        self.processors = {}
        self.configs = {} 
        self.load_all_processors()

    def load_model(self, model_name: str) -> bool:
        """FIX: Dynamically load a specific model on-demand (Zero-Touch support)"""
        folder_path = os.path.join(self.models_dir, model_name)
        config_path = os.path.join(folder_path, "model_config.json")
        if not os.path.exists(config_path):
            return False
            
        try:
            with open(config_path, "r") as f:
                config = json.load(f)
            self.configs[model_name] = config 
            
            proc_info = config.get("processor")
            if proc_info:
                file_path = os.path.join(folder_path, proc_info["file"])
                class_name = proc_info["class_name"]
                spec = importlib.util.spec_from_file_location(class_name, file_path)
                module = importlib.util.module_from_spec(spec)
                spec.loader.exec_module(module)
                ProcessorClass = getattr(module, class_name)
                self.processors[model_name] = ProcessorClass()
                logger.info(f"Zero-Touch: Dynamically loaded model {model_name} into VRAM.")
                return True
        except Exception as e:
            logger.error(f"Failed to dynamically load {model_name}: {e}")
        return False

    def load_all_processors(self):
        if not os.path.exists(self.models_dir): return
        for folder in os.listdir(self.models_dir):
            if not folder.startswith(".tmp_"):
                self.load_model(folder)

    def run_inference(self, requests: list) -> list[dict]:
        if not requests: return []
        model_name = requests[0].model_name
        
        # FIX: On-demand loading
        if model_name not in self.processors:
            success = self.load_model(model_name)
            if not success:
                return [{"status": "error", "error": f"Model {model_name} not found"} for _ in requests]
                
        processor = self.processors[model_name]
        has_batch_support = getattr(processor, "supports_batching", False)
        
        # FIX: True vectorized execution
        try:
            if has_batch_support:
                payloads = [req.payload for req in requests]
                return processor.execute_batch(payloads)
            else:
                # Fallback for models that explicitly don't support batching
                results = []
                for req in requests:
                    results.append(processor.execute(req.payload))
                return results
        except Exception as e:
            logger.error(f"Inference crash on {model_name}: {e}")
            raise

class InferenceRequest:
    def __init__(self, raw_data: str, parsed_data: dict, registry: DynamicModelRegistry):
        self.request_id = parsed_data["request_id"]
        self.model_name = parsed_data.get("model_name", "dummy_model")
        self.priority = parsed_data["priority"]
        self.created_at = parsed_data["created_at"]
        self.payload = parsed_data.get("payload")
        self.raw_data = raw_data  
        
        model_config = registry.configs.get(self.model_name, {})
        slas = model_config.get("slas", {}).get("priority_timeouts_s", {})
        sla_timeout = slas.get(self.priority, 1.0) 
        self.deadline = self.created_at + sla_timeout

class Batch:
    def __init__(self, model_name: str):
        self.model_name = model_name
        self.requests: list[InferenceRequest] = []
        self.flush_reason = None
        self.batch_start_time = time.time()  # FIX: Track actual batch formation time

class ReliableTaskFetcher:
    def __init__(self, redis, worker_id: str, registry: DynamicModelRegistry):
        self.redis = redis
        self.worker_id = worker_id
        self.processing_queue = f"queue:processing:{self.worker_id}"
        self.registry = registry

    async def fetch_task(self, source_queue: str) -> InferenceRequest | None:
        try:
            raw_payload = await self.redis.lmove(source_queue, self.processing_queue, "RIGHT", "LEFT")
            if not raw_payload: return None
            payload_str = raw_payload.decode("utf-8") if isinstance(raw_payload, bytes) else raw_payload
            return InferenceRequest(raw_data=payload_str, parsed_data=json.loads(payload_str), registry=self.registry)
        except Exception as e:
            return None

    async def complete_task(self, raw_data_str: str) -> None:
        await self.redis.lrem(self.processing_queue, 1, raw_data_str)

    async def requeue_task(self, req: InferenceRequest) -> None:
        try:
            await self.redis.lpush(f"queue:{req.priority}", req.raw_data)
            await self.complete_task(req.raw_data)
            REQUEUED_TASKS.labels(worker_id=self.worker_id).inc() 
        except Exception:
            pass

class WRRScheduler:
    def __init__(self, redis_client, fetcher: ReliableTaskFetcher):
        self.redis = redis_client
        self.fetcher = fetcher

    async def calculate_effective_weights(self) -> dict[str, int]:
        weights = settings.BASE_WEIGHTS.copy()
        current_time = time.time()
        pipe = self.redis.pipeline()
        pipe.lrange("queue:standard", -1, -1)
        pipe.lrange("queue:background", -1, -1)
        std_oldest, bg_oldest = await pipe.execute()
        
        if std_oldest:
            if current_time - json.loads(std_oldest[0])["created_at"] > 0.1:
                weights["standard"] = 5
        if bg_oldest:
            wait_time = current_time - json.loads(bg_oldest[0])["created_at"]
            if wait_time > 30.0: weights["background"] = 3
            if wait_time > 60.0: weights["background"] = 5
        return weights

    async def fetch_wrr(self) -> list[InferenceRequest]:
        weights = await self.calculate_effective_weights()
        fetched = []
        for queue_name, count in weights.items():
            for _ in range(count):
                task = await self.fetcher.fetch_task(f"queue:{queue_name}")
                if task: fetched.append(task)
                else: break 
        return fetched

class BatchBuilder:
    def __init__(self, scheduler: WRRScheduler, registry: DynamicModelRegistry):
        self.scheduler = scheduler
        self.registry = registry
        # FIX: Bucket requests by model to prevent Head-of-Line blocking
        self.model_buffers = defaultdict(list)
        self.active_batches = {}

    async def build_next_batch(self, shutdown_event: asyncio.Event) -> Batch | None:
        while not shutdown_event.is_set():
            # 1. Fetch new tasks and bucket them by model
            fetched = await self.scheduler.fetch_wrr()
            for req in fetched:
                self.model_buffers[req.model_name].append(req)
                if req.model_name not in self.active_batches:
                    self.active_batches[req.model_name] = Batch(req.model_name)

            now = time.time()

            # 2. Check all active batches for flush conditions
            for model_name, batch in list(self.active_batches.items()):
                buffer = self.model_buffers[model_name]
                
                model_config = self.registry.configs.get(model_name, {})
                max_batch_size = model_config.get("batching", {}).get("max_batch_size", settings.DEFAULT_MAX_BATCH_SIZE)
                batch_timeout_ms = model_config.get("batching", {}).get("batch_timeout_ms", settings.DEFAULT_BATCH_TIMEOUT_MS)

                # Move items from buffer to batch
                while buffer and len(batch.requests) < max_batch_size:
                    batch.requests.append(buffer.pop(0))
                
                if not batch.requests:
                    del self.active_batches[model_name]
                    continue

                # Trigger 1: Max Size
                if len(batch.requests) >= max_batch_size:
                    batch.flush_reason = "MAX_SIZE"
                    del self.active_batches[model_name]
                    return batch

                # Trigger 2: SLA Deadline Danger
                earliest_deadline = min(req.deadline for req in batch.requests)
                if (earliest_deadline - now) <= 0.05: # Give 50ms buffer to execute
                    batch.flush_reason = "EARLIEST_DEADLINE"
                    del self.active_batches[model_name]
                    return batch
                
                # Trigger 3: Batch Timeout (FIXED: Measured from batch creation, not ingestion)
                if (now - batch.batch_start_time) * 1000 >= batch_timeout_ms:
                    batch.flush_reason = "BATCH_TIMEOUT"
                    del self.active_batches[model_name]
                    return batch

            try:
                await asyncio.sleep(0.01) 
            except asyncio.CancelledError:
                break
                
        # Flush everything on shutdown
        for model_name, batch in self.active_batches.items():
            if batch.requests:
                batch.flush_reason = "SHUTDOWN"
                return batch
        return None

class InferenceWorker:
    def __init__(self, processor_registry, redis_client, fetcher: ReliableTaskFetcher):
        self.processor_registry = processor_registry
        self.redis = redis_client
        self.fetcher = fetcher
        self.circuit_breaker = CircuitBreaker() 

    def log_metrics(self, batch: Batch, inference_ms: float):
        counts = {"premium": 0, "standard": 0, "background": 0}
        wait_time_s = 0.0
        
        for req in batch.requests:
            counts[req.priority] += 1
            
        AEROSERVE_BATCH_SIZE.observe(len(batch.requests))
        if batch.flush_reason:
            AEROSERVE_BATCH_TRIGGER.labels(worker_id=WORKER_ID, reason=batch.flush_reason.lower()).inc()
            
        if batch.requests:
            wait_time_s = time.time() - min(req.created_at for req in batch.requests)
            AEROSERVE_BATCH_WAIT_TIME.observe(wait_time_s)

        INFERENCE_TIME.labels(worker_id=WORKER_ID, model_name=batch.model_name).observe(inference_ms)
        logger.info(f"Batch flushed [{batch.model_name}] | Reason: {batch.flush_reason} | Size: {len(batch.requests)} | P:{counts['premium']} S:{counts['standard']} B:{counts['background']} | Inference: {inference_ms:.1f}ms")

    async def execute(self, batch: Batch):
        if not batch.requests: return
            
        if not self.circuit_breaker.can_execute():
            logger.warning("Worker unhealthy. Requeueing batch.")
            for req in batch.requests: await self.fetcher.requeue_task(req)
            return
            
        start_time = time.monotonic()
        
        try:
            results = await asyncio.to_thread(self.processor_registry.run_inference, batch.requests)
            
            has_success = any(res.get("status") != "error" for res in results) if results else False
            if has_success: self.circuit_breaker.record_success()
            else: self.circuit_breaker.record_failure()
            
            for req, result_data in zip(batch.requests, results):
                response_payload = {"request_id": req.request_id, "status": "success" if result_data.get("status") != "error" else "error", "data": result_data}
                await self.redis.publish(f"response:{req.request_id}", json.dumps(response_payload))
                await self.fetcher.complete_task(req.raw_data)
                
        except Exception as e:
            self.circuit_breaker.record_failure()
            logger.error(f"Fatal inference crash: {e}")
            for req in batch.requests:
                error_payload = {"request_id": req.request_id, "status": "error", "error": str(e)}
                await self.redis.publish(f"response:{req.request_id}", json.dumps(error_payload))
                await self.fetcher.complete_task(req.raw_data) # Remove from processing queue since we answered the client
        finally:
            self.log_metrics(batch, (time.monotonic() - start_time) * 1000)

async def main():
    start_http_server(8002)
    
    import redis.asyncio as redis_async
    # FIX: Use REDIS_URL mapping from Phase 2
    redis_client = redis_async.Redis.from_url(settings.REDIS_URL, decode_responses=True)
    
    heartbeat = HeartbeatEmitter(redis_client, worker_id=WORKER_ID)
    await heartbeat.start()
    
    registry = DynamicModelRegistry()
    task_fetcher = ReliableTaskFetcher(redis_client, worker_id=WORKER_ID, registry=registry)
    scheduler = WRRScheduler(redis_client, task_fetcher)
    batch_builder = BatchBuilder(scheduler, registry=registry)
    worker = InferenceWorker(registry, redis_client, task_fetcher)
    
    shutdown_event = asyncio.Event()

    def handle_shutdown(sig, frame):
        logger.warning("Shutdown signal received. Finishing active batches...")
        shutdown_event.set()

    signal.signal(signal.SIGINT, handle_shutdown)
    signal.signal(signal.SIGTERM, handle_shutdown)

    logger.info(f"Worker {WORKER_ID} started. Listening for requests...")
    
    while not shutdown_event.is_set():
        batch = await batch_builder.build_next_batch(shutdown_event)
        if batch and batch.requests:
            await worker.execute(batch)
            
    await heartbeat.stop()
    await redis_client.aclose()
    logger.info("Worker shut down cleanly.")

if __name__ == "__main__":
    asyncio.run(main())

logger = logging.getLogger("TestClient")

API_URL = "http://localhost:8000/predict"

async def send_request(client: httpx.AsyncClient, req_id: int, priority: str):
    """Sends a single request and measures the exact round-trip latency."""
    payload = {
        "model_name": "test_model",
        "priority": priority,
        "payload": {"test_id": req_id}
    }
    start_time = time.time()
    
    try:
        response = await client.post(API_URL, json=payload, timeout=12.0)
        latency = (time.time() - start_time) * 1000
        
        if response.status_code == 200:
            logger.info(f"✅ [Req {req_id:03d} | {priority[:4].upper()}] Success in {latency:.0f}ms")
            return {"status": "success", "latency": latency, "priority": priority}
        else:
            logger.error(f"❌ [Req {req_id:03d} | {priority[:4].upper()}] Failed: HTTP {response.status_code}")
            return {"status": "error", "code": response.status_code}
            
    except httpx.ReadTimeout:
        logger.error(f"❌ [Req {req_id:03d} | {priority[:4].upper()}] Gateway Timeout (Client Side)")
        return {"status": "timeout"}

async def run_scenario_1_basic(client):
    logger.info("--- SCENARIO 1: Basic End-to-End ---")
    await send_request(client, 1, "premium")
    await asyncio.sleep(1)

async def run_scenario_2_mixed_batch(client):
    logger.info("\n--- SCENARIO 2: Mixed Batch (High-Water Mark Timeout) ---")
    # Send 1 Premium, 2 Standard, 2 Background simultaneously.
    # The worker should group them into ONE batch of 5.
    # The whole batch must execute in <50ms because of the Premium request!
    tasks = [
        send_request(client, 10, "premium"),
        send_request(client, 11, "standard"),
        send_request(client, 12, "standard"),
        send_request(client, 13, "background"),
        send_request(client, 14, "background")
    ]
    await asyncio.gather(*tasks)
    await asyncio.sleep(1)

async def run_scenario_3_max_batch_overflow(client):
    logger.info("\n--- SCENARIO 3: MAX_BATCH_SIZE (Overflow) ---")
    # MAX_BATCH_SIZE is 16. We will send 20 requests at the exact same time.
    # We should see the worker process one batch of 16, then a batch of 4.
    tasks = [send_request(client, 20 + i, "standard") for i in range(20)]
    results = await asyncio.gather(*tasks)
    
    successes = sum(1 for r in results if r["status"] == "success")
    logger.info(f"Result: {successes}/20 requests processed successfully.")
    await asyncio.sleep(1)

async def run_scenario_4_wrr_fairness(client):
    logger.info("\n--- SCENARIO 4: WRR Fairness Under Heavy Load ---")
    # Flood the system with 30 Premium requests, but sneak in 5 Background requests.
    # Even though Premium is flooded, the 5:3:1 WRR should ensure Background requests
    # get processed WITHOUT having to wait for all 30 Premium requests to finish.
    tasks = []
    for i in range(30):
        tasks.append(send_request(client, 100 + i, "premium"))
    for i in range(5):
        tasks.append(send_request(client, 200 + i, "background"))
        
    await asyncio.gather(*tasks)

async def main():
    async with httpx.AsyncClient() as client:
        await run_scenario_1_basic(client)
        await run_scenario_2_mixed_batch(client)
        await run_scenario_3_max_batch_overflow(client)
        await run_scenario_4_wrr_fairness(client)

if __name__ == "__main__":
    asyncio.run(main())
